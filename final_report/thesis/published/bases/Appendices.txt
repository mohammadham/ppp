import torch
import hashlib
import numpy as np

def extract_roi_and_generate_key(unet_model, medical_image_tensor):
    """
    ورودی: تصویر پزشکی به صورت تینسر Torch
    خروجی: ماسک قطعه‌بندی ROI و کلید هَش ۲۵۶ بیتی K_hash
    """
    unet_model.eval()
    with torch.no_grad():
        roi_mask = unet_model(medical_image_tensor)
        roi_mask = (roi_mask > 0.5).float()
    
    # استخراج گشتاورهای آماری ناحیه ROI
    roi_pixels = medical_image_tensor * roi_mask
    mean_val = torch.mean(roi_pixels).item()
    std_val = torch.std(roi_pixels).item()
    var_val = torch.var(roi_pixels).item()
    
    # ترکیب گشتاورها و تولید هَش SHA-256
    stat_str = f"{mean_val:.8f}_{std_val:.8f}_{var_val:.8f}"
    k_hash_hex = hashlib.sha256(stat_str.encode('utf-8')).hexdigest()
    k_hash_bits = bin(int(k_hash_hex, 16))[2:].zfill(256)
    
    return roi_mask, k_hash_hex, k_hash_bits


۲. تابع سیستم ۵بعدی ابرآشوبی (Runge-Kutta 4th Order):

def hyperchaos_5d_system(state, a=10.0, b=8/3, c=28.0, d=3.0, e=0.2):
    x, y, z, u, v = state
    dx = a * (y - x) + u
    dy = c * x - x * z + d * y + v
    dz = x * y - b * z
    du = -y * z + e * u
    dv = x * z - e * v
    return np.array([dx, dy, dz, du, dv])

def generate_5d_chaotic_sequences(k_hash_hex, num_samples):
    # تبدیل کلید هَش ۲۵۶ بیتی به شرایط اولیه سیستم
    init_x = int(k_hash_hex[0:12], 16) / (16**12)
    init_y = int(k_hash_hex[12:24], 16) / (16**12)
    init_z = int(k_hash_hex[24:36], 16) / (16**12)
    init_u = int(k_hash_hex[36:48], 16) / (16**12)
    init_v = int(k_hash_hex[48:60], 16) / (16**12)
    
    state = np.array([init_x, init_y, init_z, init_u, init_v])
    dt = 0.001
    seqs = np.zeros((num_samples, 5))
    
    # حل عددی به روش Runge-Kutta مرتبه ۴
    for i in range(num_samples):
        k1 = hyperchaos_5d_system(state)
        k2 = hyperchaos_5d_system(state + 0.5 * dt * k1)
        k3 = hyperchaos_5d_system(state + 0.5 * dt * k2)
        k4 = hyperchaos_5d_system(state + dt * k3)
        state = state + (dt / 6.0) * (k1 + 2*k2 + 2*k3 + k4)
        seqs[i] = state
        
    return seqs

def zigzag_permutation(image_matrix):
    h, w = image_matrix.shape
    result = np.zeros_like(image_matrix)
    # الگوریتم جایگشت متقاطع زیگ‌زاگی
    indices = []
    for i in range(h + w - 1):
        if i % 2 == 0:
            x = i if i < h else h - 1
            y = 0 if i < h else i - h + 1
            while x >= 0 and y < w:
                indices.append((x, y))
                x -= 1
                y += 1
        else:
            x = 0 if i < w else i - w + 1
            y = i if i < w else w - 1
            while x < h and y >= 0:
                indices.append((x, y))
                x += 1
                y -= 1
    flat_img = image_matrix.flatten()
    scrambled = np.zeros_like(flat_img)
    for new_idx, (r, c) in enumerate(indices):
        scrambled[new_idx] = image_matrix[r, c]
    return scrambled.reshape(h, w)

import zlib

def embed_metadata_saliency(image_matrix, metadata_bytes, non_roi_mask):
    # فشرده‌سازی zlib فراداده
    compressed_meta = zlib.compress(metadata_bytes, level=9)
    meta_bits = ''.join(format(byte, '08b') for byte in compressed_meta)
    
    stego_img = image_matrix.copy().flatten()
    mask_flat = non_roi_mask.flatten()
    
    bit_idx = 0
    for pixel_idx in range(len(stego_img)):
        if mask_flat[pixel_idx] == 1 and bit_idx < len(meta_bits):
            # جای‌دهی ۲ بیت کم‌ارزش LSB
            val = stego_img[pixel_idx]
            val = (val & 0xFC) | int(meta_bits[bit_idx:bit_idx+2], 2)
            stego_img[pixel_idx] = val
            bit_idx += 2
            
    return stego_img.reshape(image_matrix.shape)
